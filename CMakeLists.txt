cmake_minimum_required(VERSION 3.16)
project(FinalBuildGoap VERSION 0.1.0 LANGUAGES C)
include(GNUInstallDirs)
include(CTest)
option(FBS_BUILD_TESTS "Build module tests" ON)
option(FBS_BUILD_EXAMPLES "Build the example" ON)
add_library(fbs_goap src/goap.c)
add_library(fbs::goap ALIAS fbs_goap)
set_target_properties(fbs_goap PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED YES
  C_EXTENSIONS NO POSITION_INDEPENDENT_CODE ON WINDOWS_EXPORT_ALL_SYMBOLS ON EXPORT_NAME goap)
target_include_directories(fbs_goap PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
if(NOT MSVC)
  target_link_libraries(fbs_goap PUBLIC m)
  target_compile_options(fbs_goap PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off)
endif()
if(FBS_BUILD_TESTS AND BUILD_TESTING)
  add_executable(fbs_test_goap tests/test_goap.c)
  set_target_properties(fbs_test_goap PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
  target_link_libraries(fbs_test_goap PRIVATE fbs::goap)
  if(NOT MSVC)
    target_compile_options(fbs_test_goap PRIVATE -UNDEBUG -ffp-contract=off)
  endif()
  add_test(NAME goap COMMAND fbs_test_goap)
  set_tests_properties(goap PROPERTIES WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}/tests)
endif()
if(FBS_BUILD_EXAMPLES)
  add_executable(fbs_goap_example examples/basic.c)
  set_target_properties(fbs_goap_example PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
  target_link_libraries(fbs_goap_example PRIVATE fbs::goap)
  if(BUILD_TESTING)
    add_test(NAME goap_example COMMAND fbs_goap_example)
  endif()
endif()
install(TARGETS fbs_goap EXPORT FinalBuildGoapTargets)
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(EXPORT FinalBuildGoapTargets NAMESPACE fbs::
  DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/FinalBuildGoap)
install(FILES LICENSE THIRD_PARTY_NOTICES DESTINATION ${CMAKE_INSTALL_DATADIR}/licenses/fbs-goap)
install(DIRECTORY third_party/ DESTINATION ${CMAKE_INSTALL_DATADIR}/licenses/fbs-goap/upstream OPTIONAL)
